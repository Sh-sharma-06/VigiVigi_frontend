    57→export function Canvas() {
    58→  const artifacts = useVigilant((s) => s.artifacts);
    59→  const selectedId = useVigilant((s) => s.selectedId);
    60→  const allow = useVigilant((s) => s.allowlisted);
    61→  const scanRuns = useVigilant((s) => s.scanRuns);
    62→  const setIntakeOpen = useVigilant((s) => s.setIntakeOpen);
    63→
    64→  const a = useMemo(
    65→    () => artifacts.find((x) => x.id === selectedId) ?? null,
    66→    [artifacts, selectedId]
    67→  );
    68→
    69→  const [codeOpen, setCodeOpen] = useState(false);
    70→  const [drawerOpen, setDrawerOpen] = useState(false);
    71→
    72→  if (!a) return <EmptyState onUpload={() => setIntakeOpen(true)} />;
    73→
    74→  const eff = effectiveStatus(a, allow);
    75→  const run = scanRuns[a.id];
    76→  const scanning = eff === "scanning" || eff === "queued";
    77→
    78→  return (
    79→    <div className="flex min-h-[64vh] w-full min-w-0 flex-col">
    80→      {scanning ? (
    81→        <ScanState key={a.id} a={a} run={run} />
    82→      ) : (
    83→        <VerdictView
    84→          key={a.id}
    85→          a={a}
    86→          eff={eff}
    87→          allow={allow}
    88→          onCode={() => setCodeOpen(true)}
    89→          onDrawer={() => setDrawerOpen(true)}
    90→        />
    91→      )}
    92→
    93→      <SafeLoaderModal open={codeOpen} onOpenChange={setCodeOpen} a={a} />
    94→      <EvidenceDrawer open={drawerOpen} onOpenChange={setDrawerOpen} a={a} />
    95→    </div>
    96→  );
    97→}

   111→
   112→function toneIcon(tone: Tone, eff: EffectiveStatus) {
   113→  const cls = "h-4 w-4";
   114→  if (eff === "scanning") return <Loader2 size={15} className={`${cls} animate-spin`} />;
   115→  if (eff === "allowlisted") return <BadgeCheck size={15} className={cls} />;
   116→  switch (tone) {
   117→    case "red":
   118→      return <ShieldX size={15} className={cls} />;
   119→    case "amber":
   120→      return <ShieldAlert size={15} className={cls} />;
   121→    default:
   122→      return <ShieldCheck size={15} className={cls} />;
   123→  }
   124→}
   125→
   126→function verdictWord(eff: EffectiveStatus): { word: string; dot: string } {
   127→  switch (eff) {
   128→    case "malicious":
   129→      return { word: "Blocked", dot: "bg-[#fb5069] shadow-[0_0_8px_rgba(251,80,105,0.8)]" };
   130→    case "review":
   131→      return { word: "Needs review", dot: "bg-amber-300 shadow-[0_0_8px_rgba(251,191,36,0.8)]" };
   132→    case "allowlisted":
   133→      return { word: "Allowlisted", dot: "bg-emerald-400 shadow-[0_0_8px_rgba(52,211,153,0.8)]" };
   134→    case "clean":
   135→      return { word: "Safe", dot: "bg-emerald-400 shadow-[0_0_8px_rgba(52,211,153,0.8)]" };
   136→    case "error":
   137→      return { word: "Failed", dot: "bg-[#fb5069] shadow-[0_0_8px_rgba(251,80,105,0.8)]" };
   138→    default:
   139→      return { word: "Pending", dot: "bg-[#c4b5fd]" };
   140→  }
   141→}

   182→  const summary = plainSummary(a, allow);
   183→  const footnote = summaryFootnote(a, allow);
   184→  const allowRec = allow[a.id] as { expiry?: string } | undefined;
   185→  const isReview = eff === "review";
   186→  const isError = eff === "error";
   187→  const retryExtended = useVigilant((s) => s.retryExtended);
   188→
   189→  return (
   190→    <div className="animate-fade-in mx-auto w-full max-w-5xl px-1 py-2 lg:px-6 lg:py-6">
   191→      {/* ——— A. Hero verdict banner ——— */}
   192→      <div className="glass relative overflow-hidden rounded-3xl p-7 lg:p-8">
   193→        <div
   194→          aria-hidden

   198→          aria-hidden
   199→          className="pointer-events-none absolute -bottom-36 -left-20 h-64 w-64 rounded-full bg-[#c4b5fd]/10 blur-[70px]"

   201→
   202→        <div className="relative flex flex-wrap items-center gap-2.5">
   203→          <span
   204→            className={`inline-flex items-center gap-2 rounded-full border px-3.5 py-1.5 text-[13px] font-semibold backdrop-blur-sm ${toneBadge[meta.tone]}`}
   205→          >
   206→            {toneIcon(meta.tone, eff)}

   227→          <button
   228→            onClick={() => copySha(a.sha256)}
   229→            className="-mx-1 inline-flex items-center gap-1.5 rounded-md px-1.5 py-0.5 font-mono transition-colors duration-200 hover:bg-white/[0.09] hover:text-[#f3effc]"

   231→            title={a.sha256}

   243→            <HardDrive size={13} />
   244→            {fmtSize(a.sizeMB)}

   246→          <span aria-hidden>·</span>

   289→      </section>

   291→      {/* ——— Phase-wise data + crazy replay ——— */}

   294→      {/* ——— C. Primary action row ——— */}

   298→            onClick={onCode}

   301→            <Code2 size={16} />
   302→            View Safe-Loader Code

   315→        <Button
   316→          variant="ghost"
   317→          onClick={onDrawer}
   318→          className="h-11 gap-2 rounded-xl border border-transparent px-5 text-sm font-medium text-[#f3effc]/60 transition-all duration-200 hover:border-white/[0.12] hover:bg-white/[0.06] hover:text-[#f3effc]"

   334→      {/* ——— Remediation ——— */}

   336→
   337→      <p className="mt-6 flex items-center gap-2 text-[12.5px] text-[#f3effc]/40">
   338→        <Lock size={12.5} />
   339→        Deep technical data — opcodes, manifests, raw JSON — stays hidden until you
   340→        open it.
   341→      </p>
   342→    </div>
   343→  );
   344→}

   346→/* ------------------------------------------------------------------ */
   347→/* Stat strip — risk · confidence · scan time · decided by             */
   348→/* ------------------------------------------------------------------ */
   349→
   350→function StatStrip({ a, eff }: { a: Artifact; eff: EffectiveStatus }) {
   351→  const { word, dot } = verdictWord(eff);
   352→  const conf = CONF_PCT[a.confidence] ?? 60;
   353→  const confColor =
   354→    a.confidence === "HIGH"
   355→      ? "from-emerald-400 to-emerald-300 shadow-[0_0_12px_rgba(52,211,153,0.6)]"
   356→      : a.confidence === "MEDIUM"
   357→        ? "from-amber-400 to-amber-300 shadow-[0_0_12px_rgba(251,191,36,0.55)]"
   358→        : "from-[#a78bfa] to-[#c4b5fd] shadow-[0_0_12px_rgba(167,139,250,0.55)]";
   359→
   360→  const tiles: { label: string; node: React.ReactNode; caption: string }[] = [
   361→    {
   362→      label: "Risk level",
   363→      node: (
   364→        <span className="flex items-center gap-2 text-[15px] font-semibold text-[#f3effc]">
   365→          <span className={`h-2.5 w-2.5 rounded-full ${dot}`} />
   366→          {word}
   367→        </span>
   368→      ),
   369→      caption: a.containsExecutableCode ? "executable code present" : "no executable code",
   370→    },
   371→    {
   372→      label: "Confidence",
   373→      node: (
   374→        <span className="flex w-full items-center gap-2.5">
   375→          <span className="text-[15px] font-semibold text-[#f3effc]">{a.confidence}</span>
   376→          <span className="h-1.5 min-w-0 flex-1 overflow-hidden rounded-full bg-white/[0.09]">
   377→            <span
   378→              className={`vil-meter block h-full rounded-full bg-gradient-to-r ${confColor}`}
   379→              style={{ width: `${conf}%` }}
   380→            />
   381→          </span>
   382→        </span>
   383→      ),
   384→      caption: "verdict certainty",
   385→    },
   386→    {
   387→      label: "Scan time",
   388→      node: (
   389→        <span className="text-[15px] font-semibold text-[#f3effc]">
   390→          {fmtDuration(a.latencyMs)}
   391→        </span>
   392→      ),
   393→      caption: a.cacheHit ? "served from verdict cache" : "full 5-phase pipeline",
   394→    },
   395→    {
   396→      label: "Decided by",
   397→      node: (
   398→        <span className="truncate text-[15px] font-semibold text-[#f3effc]">
   399→          {DECIDER_LABEL[a.decidedBy] ?? a.decidedBy}
   400→        </span>
   401→      ),
   402→      caption: `${a.format} lane`,
   403→    },
   404→  ];
   405→
   406→  return (
   407→    <div className="mt-4 grid grid-cols-2 gap-3 xl:grid-cols-4" aria-label="Scan statistics">
   408→      {tiles.map((t) => (
   409→        <div key={t.label} className="glass-soft rounded-2xl px-4 py-3.5 transition-colors duration-200 hover:bg-white/[0.08]">
   410→          <p className="flex items-center gap-1.5 text-[10.5px] font-semibold uppercase tracking-[0.14em] text-[#f3effc]/45">
   411→            {t.label === "Confidence" && <Gauge size={11} className="text-[#c4b5fd]/70" />}
   412→            {t.label}
   413→          </p>
   414→          <div className="mt-2">{t.node}</div>
   415→          <p className="mt-1.5 truncate text-[11px] text-[#f3effc]/40">{t.caption}</p>
   416→        </div>
   417→      ))}
   418→    </div>
   419→  );
   420→}

   422→/* ------------------------------------------------------------------ */
   423→/* Findings panel — clickable signals that open the evidence drawer    */
   424→/* ------------------------------------------------------------------ */
   425→
   426→function FindingsPanel({ a, onDrawer }: { a: Artifact; onDrawer: () => void }) {
   427→  const blocked = a.findings.filter((f) => f.state === "BLOCKED").length;
   428→  const review = a.findings.filter((f) => f.state === "NEEDS REVIEW").length;
   429→  const info = a.findings.filter((f) => f.state === "INFO").length;
   430→
   431→  return (
   432→    <section className="glass rounded-3xl p-5 lg:p-6" aria-label="Signals found in this file">
   433→      <div className="flex flex-wrap items-center gap-2">
   434→        <h3 className="text-[15px] font-semibold tracking-tight text-[#f3effc]">
   435→          Signals in this file
   436→        </h3>
   437→        <span className="glass-soft rounded-full px-2 py-0.5 text-[11px] font-medium text-[#f3effc]/60">
   438→          {a.findings.length}
   439→        </span>
   440→        <span className="ml-auto flex items-center gap-2.5 text-[11px] text-[#f3effc]/45">
   441→          {blocked > 0 && <span className="text-[#ff92a5]">{blocked} blocked</span>}
   442→          {review > 0 && <span className="text-amber-200/90">{review} review</span>}
   443→          {info > 0 && <span>{info} info</span>}
   444→        </span>
   445→      </div>
   446→
   447→      <div className="mt-3.5 flex flex-col gap-2">
   448→        {a.findings.map((f) => (
   449→          <button
   450→            key={f.id}
   451→            onClick={onDrawer}
   452→            className="group flex w-full items-start gap-3 rounded-xl border border-transparent px-3 py-2.5 text-left transition-all duration-200 hover:border-white/[0.12] hover:bg-white/[0.05]"
   453→            title="Open in technical evidence"
   454→          >
   455→            <span
   456→              className={`mt-0.5 shrink-0 rounded-md border px-1.5 py-0.5 text-[9.5px] font-bold tracking-wide ${findingChip(f.state)}`}
   457→            >
   458→              {f.state}
   459→            </span>
   460→            <span className="min-w-0 flex-1">
   461→              <span className="flex items-baseline gap-2">
   462→                <span className="truncate font-mono text-[12px] font-semibold text-[#e9e2fb]">
   463→                  {f.location}
   464→                </span>
   465→                {f.offset && f.offset !== "—" && (
   466→                  <span className="shrink-0 font-mono text-[10.5px] text-[#f3effc]/40">
   467→                    @ {f.offset}
   468→                  </span>
   469→                )}
   470→              </span>
   471→              <span className="mt-0.5 block line-clamp-2 text-[12.5px] leading-5 text-[#f3effc]/60">
   472→                {f.evidence}
   473→              </span>
   474→            </span>
   475→            <FileSearch
   476→              size={14}
   477→              className="mt-1 shrink-0 text-[#f3effc]/25 transition-colors duration-200 group-hover:text-[#c4b5fd]"
   478→            />
   479→          </button>
   480→        ))}
   481→      </div>
   482→
   483→      <button
   484→        onClick={onDrawer}
   485→        className="mt-3.5 inline-flex items-center gap-1.5 text-[12.5px] font-medium text-[#c4b5fd]/85 transition-colors duration-200 hover:text-[#c4b5fd]"
   486→      >
   487→        Open full technical evidence
   488→        <span aria-hidden>→</span>
   489→      </button>
   490→    </section>
   491→  );
   492→}
