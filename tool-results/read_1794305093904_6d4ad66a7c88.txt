  1217→export function guessFormat(name: string): ModelFormat {
  1218→  const n = name.toLowerCase();
  1219→  if (n.endsWith(".pkl") || n.endsWith(".pickle")) return "PICKLE";
  1220→  if (n.endsWith(".keras")) return "KERAS";
  1221→  if (n.endsWith(".hdf5") || n.endsWith(".h5")) return "HDF5";
  1222→  if (n.endsWith(".gguf")) return "GGUF";
  1223→  if (n.endsWith(".onnx")) return "ONNX";
  1224→  if (n.endsWith(".safetensors")) return "SAFETENSORS";
  1225→  if (n.endsWith(".pt") || n.endsWith(".ckpt") || n.endsWith(".pth") || n.endsWith(".zip"))
  1226→    return "PYTORCH_ZIP";
  1227→  return "PICKLE";
  1228→}

  1230→export function buildUploadArtifact(spec: {
  1231→  name: string;
  1232→  format: ModelFormat;
  1233→  sizeMB: number;
  1234→  sha256: string;
  1235→  verdict: BaseVerdict;
  1236→  confidence?: "HIGH" | "MEDIUM" | "LOW";
  1237→  cacheHit?: boolean;
  1238→}): Artifact {
  1239→  const { name, format, sizeMB, sha256, verdict } = spec;
  1240→  const seed = sha256.charCodeAt(0) + sha256.charCodeAt(3);
  1241→  const ratio = 1 + ((seed % 9) / 10);
  1242→  const lane =
  1243→    format === "PICKLE" || format === "PYTORCH_ZIP"
  1244→      ? "PICKLE VM"
  1245→      : format === "KERAS" || format === "HDF5"
  1246→        ? "KERAS / HDF5 AST"
  1247→        : format === "GGUF"
  1248→          ? "GGUF HEADER"
  1249→          : format === "ONNX"
  1250→            ? "ONNX GRAPH"
  1251→            : "SAFETENSORS BOUNDS";
  1252→  const skipGvisor = verdict === "clean";
  1253→  const ext = "." + (name.split(".").pop() ?? "bin");
  1254→  const codeBearing = verdict !== "clean";
  1255→  return {
  1256→    id: `up-${uploadId()}`,
  1257→    name,
  1258→    format,
  1259→    ext,
  1260→    sizeMB,
  1261→    sha256,
  1262→    compressionRatio: Math.round(ratio * 100) / 100,
  1263→    scannedAt: new Date().toISOString(),
  1264→    ruleset: RULESET,
  1265→    engine: ENGINE,
  1266→    verdict,
  1267→    confidence: spec.confidence ?? "HIGH",
  1268→    decidedBy: skipGvisor ? "static_pickle_vm" : "gvisor_sandbox",
  1269→    containsExecutableCode: codeBearing,
  1270→    classification: codeBearing ? "Review Burden (Simulated Intake)" : undefined,
  1271→    rootCause:
  1272→      verdict === "clean"
  1273→        ? "Structural pass verified magic bytes, member bounds and 0 executable constructs; container is declarative by design."
  1274→        : "Intake simulation flagged code-bearing constructs; simulated verdict pending real backend wiring — connect /api/v1/enterprise/scan for authoritative results.",
  1275→    latencyMs: 0,
  1276→    cacheHit: spec.cacheHit ?? false,
  1277→    status: "queued",
