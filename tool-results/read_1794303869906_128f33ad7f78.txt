    21→function SortHeaderIcon({ k, sort }: { k: SortKey; sort: { key: SortKey; dir: 1 | -1 } }) {
    22→  if (sort.key !== k) return <ArrowUpDown size={11} className="opacity-40" />;
    23→  return sort.dir === 1 ? <ArrowUp size={11} /> : <ArrowDown size={11} />;

   167→function ComparePanel({ a, b, onClose }: { a: HistoryRow; b: HistoryRow; onClose: () => void }) {
   168→  const rows: { label: string; va: string; vb: string; diff: boolean }[] = [
   169→    { label: "Verdict", va: a.verdict.toUpperCase(), vb: b.verdict.toUpperCase(), diff: a.verdict !== b.verdict },
   170→    { label: "Format", va: a.format, vb: b.format, diff: a.format !== b.format },
   171→    { label: "Size", va: fmtSize(a.sizeMB), vb: fmtSize(b.sizeMB), diff: Math.abs(a.sizeMB - b.sizeMB) > 1 },
   172→    { label: "Latency", va: `${(a.latencyMs / 1000).toFixed(1)}s`, vb: `${(b.latencyMs / 1000).toFixed(1)}s`, diff: Math.abs(a.latencyMs - b.latencyMs) > 500 },
   173→    { label: "Scanned", va: fmtTime(a.scannedAt), vb: fmtTime(b.scannedAt), diff: a.scannedAt !== b.scannedAt },
   174→    { label: "SHA-256", va: a.sha256.slice(0, 16) + "…", vb: b.sha256.slice(0, 16) + "…", diff: a.sha256 !== b.sha256 },
   175→  ];
   176→  return (
   177→    <div className="rise-in mb-4 rounded-2xl border border-fuchsia-300/25 bg-fuchsia-500/[0.06] p-4">
   178→      <div className="mb-2 flex items-center gap-2">
   179→        <GitCompareArrows size={14} className="text-[#ef95bb]" />
   180→        <span className="text-[11.5px] font-bold text-[#f4f0f6]">Model Version Diff</span>
   181→        <button className="btn-ghost ml-auto !px-2 !py-0.5" onClick={onClose}>✕</button>
   182→      </div>
   183→      <div className="grid gap-2 md:grid-cols-2">
   184→        {[a, b].map((r, i) => (
   185→          <div key={i} className="inset-panel p-2.5">
   186→            <p className="mb-1 truncate text-[11px] font-bold text-[#f0ebf2]">{r.name}</p>
   187→            {rows.map((row) => (
   188→              <div key={row.label} className="flex justify-between gap-2 border-b border-white/[0.04] py-0.5 text-[10px] last:border-0">
   189→                <span className="text-[#837f93]">{row.label}</span>
   190→                <span className="font-mono text-[#b9b5c6]">{i === 0 ? row.va : row.vb}</span>
   191→              </div>
   192→            ))}
   193→          </div>
   194→        ))}
   195→      </div>
   196→    </div>
   197→  );
   198→}

   200→function download(name: string, content: string, mime = "application/json") {
   201→  const blob = new Blob([content], { type: mime });
   202→  const url = URL.createObjectURL(blob);
   203→  const a = document.createElement("a");
   204→  a.href = url;
   205→  a.download = name;
   206→  a.click();
   207→  setTimeout(() => URL.revokeObjectURL(url), 2000);
   208→}
